81def _task_lost(job, rq_job) -> bool: 

82 """Return whether an active Job's queue task can no longer run it.""" 

83 from core.choices import JobStatusChoices

85 if rq_job is None: 

86 # The Job commits before its queue push, so a young pending Job may have no task yet.

111@dataclass(frozen=True) 

112class ImportJobStatus: 

113 """What one import Job and its queue task say now, for a page that shows the Job.""" 

114 

115 job: Any 

116 state: str 

117 phase: str = "" 

118 processed: int = 0 

119 total: int = 0 

120 # Only for a queued Job whose task is in its queue: its place there (1 is next), and its queue's workers. 

121 position: int | None = None 

122 workers: int | None = None 

123 busy: int = 0 

124 queue_readable: bool = True 

125 

126 @property 

127 def active(self) -> bool: 

128 """Return whether the Job still waits in its queue or runs.""" 

129 return self.state in (QUEUED, RUNNING) 

130 

131 @property 

132 def queue_note(self) -> str: 

133 """Return what a page says when Redis did not answer, or "" when it did.""" 

134 return "" if self.queue_readable else QUEUE_UNREADABLE

137def import_job_status(job) -> ImportJobStatus: 

138 """Read one import Job, its queue task and its queue, without changing any of them.""" 

139 from core.choices import JobStatusChoices 

140 

141 data = job.data or {} 

142 status = ImportJobStatus( 

143 job, 

144 ENDED, 

145 phase=str(data.get("phase") or ""), 

146 processed=int(data.get("processed") or 0), 

147 total=int(data.get("total") or 0), 

148 ) 

149 if job.status not in JobStatusChoices.ENQUEUED_STATE_CHOICES: 

150 return status 

151 try: 

152 return _with_queue_evidence(status) 

153 except (RedisConnectionError, RedisTimeoutError): 

154 # An outage says nothing about the task, so the Job row alone decides the state. 

155 state = RUNNING if job.status == JobStatusChoices.STATUS_RUNNING else QUEUED 

156 return replace(status, state=state, queue_readable=False)

159def _with_queue_evidence(status: ImportJobStatus) -> ImportJobStatus: 

160 """Add what the Job's queue task and its queue say; Redis errors reach the caller.""" 

161 from core.choices import JobStatusChoices 

162 

163 job = status.job 

164 rq_job = import_queue_task(job) 

165 if _task_lost(job, rq_job): 

166 return replace(status, state=LOST) 

167 if rq_job is None: 

168 # The queue push follows the commit, so the queue knows nothing of this Job yet. 

169 return replace(status, state=QUEUED) 

170 # The worker publishes row progress to the task, not to the Job row. 

171 status = replace( 

172 status, 

173 phase=str(rq_job.meta.get("phase") or status.phase), 

174 processed=int(rq_job.meta.get("processed") or status.processed), 

175 total=int(rq_job.meta.get("total") or status.total), 

176 ) 

177 if job.status == JobStatusChoices.STATUS_RUNNING: 

178 return replace(status, state=RUNNING) 

179 return _with_queue_place(replace(status, state=QUEUED), rq_job)

182def _with_queue_place(status: ImportJobStatus, rq_job) -> ImportJobStatus: 

183 """Add the task's place in its own queue, and the live workers of that queue; a read that writes nothing.""" 

184 import django_rq 

185 

186 queue = django_rq.get_queue(status.job.queue_name) 

187 # Worker.all() removes the registration of an expired worker, so read each worker hash directly. 

188 states = [queue.connection.hget(key, "state") for key in Worker.all_keys(queue=queue)] 

189 live = [as_text(state) for state in states if state is not None] 

190 position = rq_job.get_position() 

191 return replace( 

192 status, 

193 position=None if position is None else position + 1, 

194 workers=len(live), 

195 busy=live.count(WorkerStatus.BUSY), 

196 )

199def _try_import_job_lock(job) -> bool: 

200 """Take the Job lock until the transaction ends, or return False at once when a worker holds it.""" 

201 if not connection.in_atomic_block: 

202 raise RuntimeError("The import Job lock requires the preview command transaction.") 

203 # Never wait for a worker while holding its profile. PostgreSQL holds this lock through commit or rollback. 

204 with connection.cursor() as cursor: 

205 cursor.execute("SELECT pg_try_advisory_xact_lock(%s)", [_import_job_lock(job)]) 

206 return cursor.fetchone()[0]

225def cancel_queued_import_job(job, message: str) -> bool: 

226 """Fail a Job that no worker has started, and cancel its queue task after the commit. 

227 

228 Return False, and change nothing, when a worker holds the Job lock or the Job has left the queue. 

229 """ 

230 from core.choices import JobStatusChoices 

231 from core.models import Job 

232 

233 if not _try_import_job_lock(job): 

234 return False 

235 job.refresh_from_db() 

236 waiting = (JobStatusChoices.STATUS_PENDING, JobStatusChoices.STATUS_SCHEDULED) 

237 if job.status not in waiting or job.started is not None: 

238 return False 

239 # NetBox has no cancelled status; a failed Job with this phase is one the operator stopped. 

240 Job.objects.filter(pk=job.pk).update( 

241 status=JobStatusChoices.STATUS_FAILED, 

242 completed=timezone.now(), 

243 error=message, 

244 data={**(job.data or {}), "phase": "cancelled", "message": message}, 

245 ) 

246 transaction.on_commit(partial(_cancel_queue_task, job)) 

247 return True

250def _cancel_queue_task(job) -> None: 

251 """Take a cancelled Job's task out of its queue; a worker that already took it skips the Job.""" 

252 try: 

253 rq_job = import_queue_task(job) 

254 if rq_job is not None and rq_job.get_status(refresh=True) in ( 

255 JobStatus.QUEUED, 

256 JobStatus.SCHEDULED, 

257 JobStatus.DEFERRED, 

258 ): 

259 rq_job.cancel() 

260 except (InvalidJobOperation, NoSuchJobError): 

261 # Another actor, such as NetBox Job.delete(), removed the task first, which is the goal. 

262 logger.debug("The queue task of cancelled import Job %s was already gone.", job.pk) 

263 except (RedisConnectionError, RedisTimeoutError): 

264 # The Job row is the record: ImportJobRunner.handle runs only a Job that is still enqueued. 

265 logger.warning("The queue task of cancelled import Job %s stays in its queue.", job.pk)

297 def _fail(self, message) -> NoReturn: 

298 """Record a recoverable failure and stop the native Job.""" 

299 values = {"phase": "failed", "message": message}

334 def run(self, profile_id, source_document_id, accepted_plan, selection, idempotency_key): 

335 """Execute one accepted Import Plan as the Job's actor.""" 

336 try: 

337 branching.fail_job_in_branch(self) 

338 except JobFailed as exc: 

339 self._save_data(phase="failed", message=str(exc)) 

340 raise 

341 user = self.job.user 

342 if user is None: 

343 self._fail("The user who started this import is no longer available.") 

344 

345 profile = ImportProfile.objects.restrict(user, "change").filter(pk=profile_id).first() 

346 if profile is None: 

347 self._fail("The import profile is no longer available.") 

348 try: 

349 validate_registered_adapter(profile) 

350 except ValidationError as exc: 

351 self._fail(operator_failure_message(exc)) 

352 source_document = SourceDocument.objects.filter(pk=source_document_id, profile=profile).first() 

353 if source_document is None: 

354 self._fail("The stored source is no longer available. Upload it again.") 

355 

356 self._save_data(phase="validating") 

357 progress = {"processed": 0, "total": 0} 

358 

359 def publish_progress(processed, total): 

360 """Remember final progress and publish the bounded RQ updates.""" 

361 progress.update(processed=processed, total=total) 

362 self._publish_progress(processed, total) 

363 

364 # Only event_tracking: the other request processors (netbox-branching) must not run in a worker. 

365 with event_tracking(self._change_logging_request(user)): 

366 try: 

367 execution = ImportEngine.execute( 

368 profile, 

369 source_document, 

370 accepted_plan, 

371 selection, 

372 idempotency_key, 

373 user, 

374 job=self.job, 

375 progress_callback=publish_progress, 

376 ) 

377 except ImportProfile.DoesNotExist: 

378 self._fail("The import profile is no longer available.") 

379 except DatabaseError as exc: 

380 logger.exception("Import execution failed with a database error") 

381 self._fail(operator_failure_message(exc)) 

382 except (EngineConfigurationError, SourceUnreadable, UnknownSourceAdapter) as exc: 

383 logger.exception("Import execution failed before its source could be planned") 

384 self._fail(operator_failure_message(exc)) 

385 except ( 

386 ObjectPermissionDenied, 

387 PlanError, 

388 PlanningTargetUnavailable, 

389 PreconditionFailed, 

390 SelectionError, 

391 StalePlan, 

392 StaleSourceDocument, 

393 ValidationError, 

394 ) as exc: 

395 self._fail(operator_failure_message(exc)) 

396 # Raising here leaves the block before event_tracking flushes the queued events. 

397 if execution.outcome != ExecutionOutcome.SUCCEEDED: 

398 reason = (execution.failure_detail or {}).get("reason") or execution.outcome or "unknown" 

399 self._fail(f"The accepted import execution did not succeed ({reason}).") 

400 self._save_data( 

401 phase="completed", 

402 processed=progress["processed"], 

403 total=progress["total"], 

404 import_execution_id=execution.pk, 

405 )
