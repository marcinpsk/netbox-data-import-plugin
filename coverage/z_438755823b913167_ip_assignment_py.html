3"""Where an address off a source row lands on a device. 

4 

5The preview names the interface and the sync writes to it, so both read this module. Two copies 

6would let a row promise one interface and the write pick another. 

7""" 

8 

9from __future__ import annotations 

10 

11import ipaddress 

12import re 

13from dataclasses import dataclass 

14from typing import Any 

15 

16# `oob_ip` carries no family in NetBox, so it takes either. 

17IP_FIELD_FAMILY: dict[str, int | None] = {"primary_ip4": 4, "primary_ip6": 6, "oob_ip": None} 

18 

19# Bounded at both ends: a word cannot leave a shorter valid address, and 45 covers the longest one. 

20_IP_TOKEN = re.compile(r"(?<![0-9A-Za-z])[0-9A-Fa-f:.]{1,45}(?:/\d{1,3})?(?![0-9A-Za-z])") 

21 

22 

23class IPAssignmentError(Exception): 

24 """The address cannot be placed, with the repair the operator has to make.""" 

25 

26 

27@dataclass(frozen=True) 

28class IPTarget: 

29 """The interface an address would land on, and the row it would use.""" 

30 

31 address: str 

32 interface: Any 

33 existing: Any | None 

34 already_held: bool 

35 

36 @property 

37 def interface_name(self) -> str: 

38 """Return the interface name, or an empty string when the row is unassigned.""" 

39 return getattr(self.interface, "name", "") or "" 

40 

41 @property 

42 def held(self): 

43 """Return the stored row the device already carries, which only a held target has.""" 

44 if not self.already_held or self.existing is None: 

45 raise IPAssignmentError(f"{self.address} is not already on this device, so it has no stored row.") 

46 return self.existing 

47 

48 @property 

49 def summary(self) -> str: 

50 """Return what the sync reports once it has written.""" 

51 if not self.interface_name: 

52 return self.address 

53 return f"{self.address} on {self.interface_name}" 

54 

55 @property 

56 def placement(self) -> str: 

57 """Return what the preview shows beside the address it already prints.""" 

58 if not self.interface_name: 

59 return "" 

60 if self.already_held: 

61 # The stored mask can differ from the one the row states, so it is worth printing. 

62 return f"already on {self.interface_name} as {self.address}" 

63 return f"would go to {self.interface_name}"

66def _normalized_ip(token: str) -> str | None: 

67 """Return *token* as 'address/prefix', or None when it is not one address.""" 

68 try: 

69 if "/" in token: 

70 return str(ipaddress.ip_interface(token)) 

71 addr = ipaddress.ip_address(token) 

72 except ValueError: 

73 return None 

74 return f"{addr}/32" if addr.version == 4 else f"{addr}/128" 

75 

76 

77def parse_address(raw_value) -> str | None: 

78 """Return the one address a source value names, as 'address/prefix', or None. 

79 

80 Sources export an address inside a label or with a separator appended, so the whole value is 

81 tried first and the addresses spelled inside it only after that. 

82 """ 

83 raw = str(raw_value).strip() 

84 if not raw: 

85 return None 

86 whole = _normalized_ip(raw) 

87 if whole is not None: 

88 return whole 

89 for token in _IP_TOKEN.findall(raw): 

90 while token: 

91 found = _normalized_ip(token) 

92 if found is not None: 

93 return found 

94 if token[-1] not in ".:": 

95 break 

96 token = token[:-1] 

97 return None

100def normalized_address(field: str, value) -> str: 

101 """Return the address the row names as 'host/prefix', in the family the field takes.""" 

102 address = parse_address(value) 

103 if address is None: 

104 raise IPAssignmentError(f"Cannot read an IP address from '{value}'.") 

105 family = IP_FIELD_FAMILY.get(field) 

106 version = ipaddress.ip_interface(address).version 

107 if family is not None and version != family: 

108 raise IPAssignmentError(f"'{value}' is an IPv{version} address; this field takes IPv{family}.") 

109 return address

112def already_assigned(device, field, address) -> bool: 

113 """Return whether the device already carries exactly this address on *field*. 

114 

115 The writer selects the address that this device holds. The primary field must point to that 

116 same row, while duplicate rows held by other objects do not affect the settled state. 

117 """ 

118 current = getattr(device, field, None) 

119 if current is None: 

120 return False 

121 try: 

122 if _host(current.address) != _host(address): 

123 return False 

124 held = held_by_device(device, address) 

125 except ValueError: 

126 return False 

127 return held is not None and held.pk == current.pk

129 

130def _host(address) -> str: 

131 """Return the host part, which is what identifies an address inside one VRF.""" 

132 return str(ipaddress.ip_interface(str(address)).ip)

135def held_by_device(device, address): 

136 """Return the address this device already carries, or None. 

137 

138 Every interface is searched, not only a management one, and the mask is ignored: a workbook 

139 states a bare address where the device holds the same host inside its real subnet. 

140 

141 An address the device points at but no interface of its own holds is not a match. NetBox 

142 requires an IP field to name an address on that device, so that state is repaired through the 

143 normal path rather than copied onto a second field. 

144 """ 

145 from ipam.models import IPAddress 

146 

147 wanted = _host(address) 

148 candidates = IPAddress.objects.filter(interface__device=device).select_related("vrf") 

149 matches = [candidate for candidate in candidates if _host(candidate.address) == wanted] 

150 if not matches: 

151 return None 

152 # A management interface answers first when the device holds the address more than once. 

153 matches.sort(key=lambda ip: not getattr(ip.assigned_object, "mgmt_only", False)) 

154 return matches[0]

157def interface_for(device): 

158 """Return the interface an address off a source file belongs on. 

159 

160 A management interface answers first: that is what the device type marks it for, and an 

161 address in a source workbook is a management address far more often than not. 

162 """ 

163 from dcim.models import InterfaceTemplate 

164 

165 interfaces = sorted(device.interfaces.all(), key=lambda i: (not i.mgmt_only, i.name)) 

166 if interfaces: 

167 return interfaces[0] 

168 model = device.device_type.model 

169 declared = list(InterfaceTemplate.objects.filter(device_type=device.device_type).order_by("name")[:5]) 

170 if not declared: 

171 raise IPAssignmentError( 

172 f"The device type '{model}' declares no interfaces, so there is nowhere to put this " 

173 f"address. Add an interface to the device type, then sync again." 

174 ) 

175 names = ", ".join(template.name for template in declared) 

176 raise IPAssignmentError( 

177 f"This device has none of the interfaces its type '{model}' declares ({names}). " 

178 f"Add them to the device, then sync again." 

179 )

182def resolve(device, field: str, value) -> IPTarget: 

183 """Return where *value* would land on *device*, or raise with the repair to make.""" 

184 from ipam.models import IPAddress 

185 

186 address = normalized_address(field, value) 

187 held = held_by_device(device, address) 

188 if held is not None: 

189 return IPTarget(address=str(held.address), interface=held.assigned_object, existing=held, already_held=True) 

190 

191 interface = interface_for(device) 

192 # The interface's VRF scopes the address: the same host in another VRF is a different address. 

193 existing = IPAddress.objects.filter(address__net_host=_host(address), vrf=interface.vrf).first() 

194 if existing is not None and existing.assigned_object is not None: 

195 owner = getattr(existing.assigned_object, "device", None) or existing.assigned_object 

196 raise IPAssignmentError(f"Address {existing.address} is already assigned to '{owner}'.") 

197 return IPTarget(address=address, interface=interface, existing=existing, already_held=False)

200def apply(target: IPTarget, user=None): 

201 """Write the address *target* names onto its interface, and return the IPAddress. 

202 

203 The row sync and the import writer both land here, so an address is created, scoped and 

204 checked the same way whichever one runs. 

205 """ 

206 from ipam.models import IPAddress 

207 

208 from .object_permissions import enforce_saved_object_permission 

209 

210 address = target.existing 

211 action = "change" if address is not None else "add" 

212 if address is None: 

213 address = IPAddress(address=target.address, vrf=target.interface.vrf) 

214 address.assigned_object = target.interface 

215 address.full_clean() 

216 address.save() 

217 # An ObjectPermission's constraints are only evaluated against the saved row. 

218 enforce_saved_object_permission(address, user, action) 

219 return address
