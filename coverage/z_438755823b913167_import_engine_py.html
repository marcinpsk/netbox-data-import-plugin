3"""The target-neutral Import Engine coordinator. 

4 

5Section 2.3 gives the coordinator the merged dependency graph, transaction scope, idempotency and

7workbook, no column and no NetBox object type. 

8"""

40 """Return one execution failure as the text the operator reads.

74class ImportEngine: 

75 """Coordinate source interpretation and Target Module planning."""

77 @classmethod 

78 def plan(cls, profile, source_document, actor, planning_context) -> ImportPlan: 

79 """Return the deterministic Import Plan for one stored source.""" 

80 document = cls._stored_source(profile, source_document) 

81 adapter = adapters.get_adapter(profile.source_adapter) 

82 if adapter is None: 

83 raise adapters.UnknownSourceAdapter( 

84 f"This release does not register the source adapter '{profile.source_adapter}'." 

85 ) 

86 if not catalog.has_implemented_module(adapter.output_kinds): 

87 raise adapters.UnknownSourceAdapter( 

88 f"This release has no Target Module for source adapter '{profile.source_adapter}'." 

89 ) 

90 source_batch = adapter.interpret( 

91 bytes(document.content), 

92 adapter_config.interpreter_config_for(profile), 

93 collect_unused=True, 

94 ) 

95 # The catalog already declares which output kinds the resolution policy applies to. 

96 if _resolution_section().applies_to(source_batch.output_kinds): 

97 source_batch = adapters.SourceBatch( 

98 output_kinds=source_batch.output_kinds, 

99 rows=tuple(derive_effective_rows(list(source_batch.rows), profile)), 

100 diagnostics=source_batch.diagnostics, 

101 unused_columns=source_batch.unused_columns, 

102 ) 

103 reader = NetBoxReader.for_actor(actor).for_planning_context(planning_context) 

104 

105 units = [] 

106 for declaration in catalog.TARGET_MODULES: 

107 if not declaration.consumes & source_batch.output_kinds: 

108 continue 

109 runtime = target_modules.runtime_for(declaration.key) 

110 if runtime is not None: 

111 units.extend(runtime.plan(source_batch, profile, catalog.CATALOG, reader)) 

112 

113 # Section 4.4 makes the merged graph the coordinator's, so a bad one fails here, not at a write. 

114 merge_changes(executable_units(units)) 

115 return ImportPlan( 

116 units=tuple(units), 

117 diagnostics=( 

118 *(cls._source_diagnostic(item) for item in source_batch.diagnostics), 

119 *( 

120 cls._unused_column_diagnostic(adapter.key, name, stats) 

121 for name, stats in source_batch.unused_columns.items() 

122 ), 

123 ), 

124 source_fingerprint=document.content_fingerprint, 

125 profile_fingerprint=profile.planning_fingerprint, 

126 actor=str(actor.pk), 

127 planning_context=planning_context, 

128 )

130 @classmethod 

131 def execute( 

132 cls, 

133 profile, 

134 source_document, 

135 accepted_plan, 

136 selection, 

137 idempotency_key, 

138 actor, 

139 *, 

140 job=None, 

141 progress_callback=None, 

142 ) -> ImportExecution: 

143 """Apply selected units from one accepted serialized plan and return their audit row.""" 

144 accepted = ImportPlan.from_dict(accepted_plan) 

145 selected_identities = tuple(selection) 

146 if not selected_identities: 

147 raise SelectionError("An execution needs at least one Synchronization Unit.") 

148 execution, created = ImportExecution.reserve( 

149 profile=profile, 

150 source_document=source_document, 

151 actor=actor, 

152 idempotency_key=idempotency_key, 

153 plan_schema_version=accepted.schema_version, 

154 accepted_plan_fingerprint=accepted.fingerprint, 

155 selected_units=list(selected_identities), 

156 input_filename=source_document.filename, 

157 ) 

158 if not created: 

159 return execution 

160 

161 try: 

162 if job is not None: 

163 execution.link_job(job) 

164 target = NetBoxReader.for_actor(actor).for_planning_context(accepted.planning_context) 

165 execution.site_name = str(target.site) 

166 execution.save(update_fields=["site_name"]) 

167 # One lock over the replan, the comparison and the writes: policy cannot move between them. 

168 with locked_profile_policy(profile.pk): 

169 profile.refresh_from_db() 

170 cls._write_selection( 

171 execution, 

172 profile, 

173 source_document, 

174 accepted, 

175 selected_identities, 

176 actor, 

177 progress_callback, 

178 ) 

179 except _ExecutionFailed as failure: 

180 cls._mark_failed( 

181 execution, 

182 reason=failure.reason, 

183 failed_change=failure.failed_change, 

184 rolled_back=failure.rolled_back, 

185 not_attempted=failure.not_attempted, 

186 ) 

187 raise failure.cause 

188 except Exception as exc: 

189 # Every other failure after the reservation, so the row can never stay pending. 

190 cls._mark_failed( 

191 execution, 

192 reason=cls._failure_reason(exc), 

193 not_attempted=cls._selected_change_identities(accepted, selected_identities), 

194 ) 

195 raise 

196 return execution

198 @classmethod 

199 def _write_selection( 

200 cls, 

201 execution, 

202 profile, 

203 source_document, 

204 accepted, 

205 selected_identities, 

206 actor, 

207 progress_callback, 

208 ) -> None: 

209 """Compare the selection against a fresh plan and apply it, inside the caller's transaction.""" 

210 current = cls.plan(profile, source_document, actor, accepted.planning_context) 

211 units = cls._selected_units(accepted, current, selected_identities) 

212 try: 

213 changes = merge_changes(units) 

214 except PlanInvalid as exc: 

215 raise SelectionError(str(exc)) from exc 

216 total = len(units) + len(changes) 

217 if progress_callback is not None: 

218 progress_callback(0, total) 

219 progress_callback(len(units), total) 

220 context = ExecutionContext( 

221 actor=actor, 

222 reader=NetBoxReader.for_actor(actor).for_planning_context(accepted.planning_context), 

223 profile=profile, 

224 ) 

225 completed: list[str] = [] 

226 for index, change in enumerate(changes): 

227 runtime = target_modules.runtime_for(change.target_module) 

228 if runtime is None: 

229 raise EngineConfigurationError(f"No Target Module runtime is registered for '{change.target_module}'.") 

230 try: 

231 runtime.apply(change, context) 

232 except (PreconditionFailed, ObjectPermissionDenied, ValidationError, DatabaseError) as exc: 

233 raise _ExecutionFailed( 

234 cause=exc, 

235 reason=cls._failure_reason(exc), 

236 failed_change=change.identity, 

237 rolled_back=completed, 

238 not_attempted=[later.identity for later in changes[index + 1 :]], 

239 ) from exc 

240 completed.append(change.identity) 

241 if progress_callback is not None: 

242 progress_callback(len(units) + index + 1, total) 

243 execution.mark_succeeded( 

244 applied_changes={"changes": completed, "deleted": []}, 

245 result_counts=cls._result_counts(changes), 

246 )

248 @staticmethod 

249 def _result_counts(changes) -> dict: 

250 """Count successful primary creates by Target Module for audit display.""" 

251 created: dict[str, int] = {} 

252 for change in changes: 

253 if change.operation == "create": 

254 created[change.target_module] = created.get(change.target_module, 0) + 1 

255 return {"created": created, "errors": 0}

257 @staticmethod 

258 def _failure_reason(exc) -> str: 

259 """Return the typed audit reason for one failure a Target Module can raise.""" 

260 if isinstance(exc, StalePlan): 

261 return FailureReason.STALE_PLAN 

262 if isinstance(exc, SelectionError): 

263 return FailureReason.SELECTION 

264 if isinstance(exc, (StaleSourceDocument, PlanningTargetUnavailable)): 

265 return FailureReason.PLANNING 

266 if isinstance(exc, PreconditionFailed): 

267 return FailureReason.PRECONDITION 

268 if isinstance(exc, ObjectPermissionDenied): 

269 return FailureReason.PERMISSION 

270 if isinstance(exc, ValidationError): 

271 return FailureReason.VALIDATION 

272 if isinstance(exc, DatabaseError): 

273 return FailureReason.DATABASE 

274 return FailureReason.PLANNING

276 @staticmethod 

277 def _selected_units(accepted, current, selected_identities): 

278 """Return the current actionable units whose accepted fingerprints still match.""" 

279 if len(set(selected_identities)) != len(selected_identities): 

280 raise SelectionError("A Synchronization Unit can be selected only once.") 

281 selected = [] 

282 for identity in selected_identities: 

283 accepted_unit = accepted.unit(identity) 

284 current_unit = current.unit(identity) 

285 if accepted_unit is None or current_unit is None: 

286 raise SelectionError(f"The current Import Plan does not carry selected unit '{identity}'.") 

287 if accepted_unit.disposition != Disposition.ACTIONABLE: 

288 raise SelectionError(f"Synchronization Unit '{identity}' was not actionable when selected.") 

289 if current_unit.disposition != Disposition.ACTIONABLE: 

290 raise StalePlan(f"Synchronization Unit '{identity}' is no longer actionable.") 

291 if accepted.unit_fingerprint(identity) != current.unit_fingerprint(identity): 

292 raise StalePlan(f"Synchronization Unit '{identity}' changed after the plan was accepted.") 

293 selected.append(current_unit) 

294 return tuple(selected)

296 @staticmethod 

297 def _selected_change_identities(plan, selected_identities) -> list[str]: 

298 """Return unique Planned Change identities carried by the selected units.""" 

299 identities = [] 

300 for unit_identity in selected_identities: 

301 unit = plan.unit(unit_identity) 

302 if unit is None: 

303 continue 

304 for change in unit.changes: 

305 if change.identity not in identities: 

306 identities.append(change.identity) 

307 return identities

309 @staticmethod 

310 def _mark_failed(execution, **detail) -> None: 

311 """Finish a failed row unless a concurrent finisher already chose its outcome.""" 

312 try: 

313 execution.mark_failed(**detail) 

314 except ValueError: 

315 pass

317 @staticmethod 

318 def _stored_source(profile, source_document) -> SourceDocument: 

319 """Return the stored bytes this profile may plan, refusing a reference it does not own.""" 

320 document = SourceDocument.objects.filter(pk=source_document.pk).first() 

321 if document is None: 

322 raise StaleSourceDocument("The stored source no longer exists. Upload it again.") 

323 if document.profile_id != profile.pk: 

324 raise StaleSourceDocument( 

325 f"Source document {document.pk} belongs to another Import Profile, so this one cannot plan it." 

326 ) 

327 return document

329 @staticmethod 

330 def _source_diagnostic(diagnostic) -> Diagnostic: 

331 """Return one adapter diagnostic in the plan vocabulary, under the adapter's own namespace. 

332 

333 The code passes through: a Source Adapter names its own domain, and prefixing a second one 

334 would push a namespaced code past what a plan diagnostic code accepts. 

335 """ 

336 display = {"message": diagnostic.message} 

337 if diagnostic.row_number is not None: 

338 display["row_number"] = diagnostic.row_number 

339 return Diagnostic(code=diagnostic.code, severity=Severity.WARNING, display=display)
